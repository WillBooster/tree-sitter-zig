fn f(opt: ?u32, res: anyerror!u32) u32 {
    if (opt) |value| {
        return value;
    } else if (res) |ok| {
        return ok;
    } else |err| {
        _ = err;
    }
    const y = if (opt == null) 1 else 2;
    return y;
}

fn g(items: []const u8) usize {
    var total: usize = 0;
    for (items, 0..) |item, index| {
        if (item == 0) continue;
        total += index;
    }
    outer: while (true) {
        inner: for (0..10) |n| {
            if (n == 5) break :outer;
            if (n == 3) continue :inner;
        }
    }
    var it: ?u8 = 1;
    while (it) |v| : (it = null) {
        _ = v;
    } else {
        total = 0;
    }
    inline for (.{ 1, 2 }) |k| total += k;
    return total;
}

    const r = switch (x) {
        0 => 1,
        1, 2 => 3,
        4...9 => |n| n,
        else => blk: {
            break :blk 0;
        },
    };
    switch (v) {
        .int => |*p| p.* += 1,
        inline .float, .none => {},
    }
    return r;
}

fn k(allocator: Allocator) !void {
    const buf = try allocator.alloc(u8, 16);
    defer allocator.free(buf);
    errdefer |err| log(err);
    const n = read(buf) catch |err| switch (err) {
        error.EndOfStream => 0,
        else => return err,
    };
    const m = maybe() orelse return;
    _ = n;
    _ = m;
    nosuspend {}
    suspend {}
    unreachable;
}
